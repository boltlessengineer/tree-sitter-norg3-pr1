================================================================================
[word]
--------------------------------------------------------------------------------
(document
  (paragraph
    (anchor
      markup: (markup
        (word)))))

================================================================================
anchor-01: Markup inside anchor description
================================================================================
[*word /word/*]
--------------------------------------------------------------------------------
(document
  (paragraph
    (anchor
      markup: (markup
        (bold
          (bold_open)
          (word)
          (whitespace)
          (italic
            (italic_open)
            (word)
            (italic_close))
          (bold_close))))))
